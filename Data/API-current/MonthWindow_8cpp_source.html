58void MonthWindow::MessageReceived(BMessage *msg)
59{
60 if(msg->what=='MVME')
61 {
62 // Is date correct?
63 int32 day, month, year;
64 if(msg->FindInt32("day",&day)!=B_OK) return;
65 if(msg->FindInt32("month",&month)!=B_OK) return;
66 if(msg->FindInt32("year",&year)!=B_OK) return;
67
68 if(year<first_year || year>last_year) return;
69 if(month<1 || month>12) return;
70 int32 tmp;
71
72 tmp=31;
73 if(month==4 || month==6 || month==9 || month==11) tmp=30;
74 else if(month==2)
75 {
76 if(year%4==0)
77 {
78 if(year%100==0 && year%400!=0) tmp=28;
79 else tmp=29;
80 }
81 else tmp=28;
82 }
83 if(day<1 || day>tmp) return;
84 msng->SendMessage(msg);
85 Quit();
86 }
87 else BWindow::MessageReceived(msg);
88}
