70DateTextView::DateTextView(int day, int month, int year,
71 uint32 flags, uint32 look)
72 :BTextView(BRect(0,0,110,15),"DateTextViewAViX",
73 BRect(0,0,110,15),B_FOLLOW_LEFT | B_FOLLOW_TOP,
74 B_WILL_DRAW | B_NAVIGABLE)
75{
76 enabled=true;
77
78 is_ins=false;
79
80 interface=look & CC_ALL_INTERFACES;
81
82 div[0]=".";
83 div[1]="/";
84 div[2]="-";
85
86 divider=look & CC_ALL_DIVIDERS;
87 if(divider>LAST_DIVIDER || divider<0) divider=0; // index of divider
88
89 this->flags=(flags & (LAST_FORMAT | CC_SHORT_YEAR | CC_HALF_CENTURY));
90 if((this->flags & (CC_SHORT_YEAR | CC_HALF_CENTURY))==CC_HALF_CENTURY)
91 this->flags=this->flags^CC_HALF_CENTURY; // XOR; CC_FULL_YEAR and CC_HALF_CENTURY
92 // at the same time may not be used
93
94 first_year=1;
95 last_year=9999;
96
97 if((this->flags & CC_SHORT_YEAR)==CC_SHORT_YEAR)
98 {
99 textlen=8;
100
101 // Changing first and last acceptable years
102 // Working in range of century defined by year variable
103
104 if(year<first_year || year>last_year)
105 {
106 // Range is set relative to today's year
107 struct tm *dat;
108 time_t tmp;
109 time(&tmp);
110 dat=localtime(&tmp);
111
112 year=dat->tm_year+1900; // today's year
113 }
114
115 first_year=(year/100)*100+1; // dividing with loosing rest
116 last_year=first_year+99;
117 if(last_year==10000) last_year--; // full century
118
119 if((this->flags & CC_HALF_CENTURY)==CC_HALF_CENTURY)
120 {
121 if(year<51 || year>9950)
122 this->flags=this->flags^CC_HALF_CENTURY; // HALF_CENTURY may nor be set
123 else
124 {
125 if((year%100)>50)
126 {
127 first_year+=50;
128 last_year+=50;
129 }
130 else
131 {
132 first_year-=50;
133 last_year-=50;
134 }
135 }
136 }
137 }
138 else textlen=10; // length of text string
139
140 SetDate(day,month,year);
141
142 float width=0;
143 BString s("");
144 for(char i='0'; i<='9'; i++)
145 {
146 s<<i;
147 if(StringWidth(s.String())>width) width=StringWidth(s.String());
148 s.SetTo("");
149 }
150
151 ResizeTo(width*(textlen-2)+StringWidth("/")*2.5, LineHeight()-1);
152
153 SetWordWrap(false);
154 SetMaxBytes(textlen);
155 SetTextRect(Bounds());
156 SetDoesUndo(false);
157 for(int32 i=0;i<256;i++) DisallowChar(i);
158 for(int32 i='0';i<='9';i++) AllowChar(i);
159}

196void DateTextView::KeyDown(const char *bytes, int32 numBytes)
197{
198 if(!enabled) if(bytes[0]!=B_TAB) return;
199
200 int32 i1,i2;
201 GetSelection(&i1,&i2);
202
203 if(bytes[0]>='0' && bytes[0]<='9')
204 {
205 if(i1>(textlen-1)) return; // not to insert after end of string
206 Select(i1,i1);
207 if(i1==2 || i1==5)
208 {
209 i1++;
210 Select(i1,i1);
211 }
212 DeleteText(i1,i1+1);
213 BTextView::KeyDown(bytes, numBytes);
214 }
215 else if(bytes[0]==B_DELETE)
216 {
217 if(i1>(textlen-1)) return; // not to insert after end of string
218 Select(i1,i1);
219 is_ins=true; // symbol "0" or divider will be inserted
220 BTextView::KeyDown(bytes, numBytes);
221 is_ins=false;
222 }
223 else if(bytes[0]==B_BACKSPACE)
224 {
225 Select(i1,i1);
226 is_ins=true;
227 BTextView::KeyDown(bytes, numBytes);
228 is_ins=false;
229 }
230 else if(bytes[0]==B_TAB)
231 {
232 Parent()->KeyDown(bytes, numBytes);
233 }
234 else if(bytes[0]==B_DOWN_ARROW)
235 {
236 // Is Ctrl+DownArrow pressed?
237 if(modifiers() & B_CONTROL_KEY)
238 {
239 // yes
240 BMessage msg(CalendarControlButtonPressedMessage);
241 Parent()->MessageReceived(&msg);
242 }
243 else
244 BTextView::KeyDown(bytes, numBytes);
245 }
246 else
247 BTextView::KeyDown(bytes, numBytes);
248}

317void DateTextView::DrawDate(int day, int month, int year)
318{
319 // It is assumed that date is correct
320 BString s;
321 s.SetTo("");
322
323 if(!((flags & CC_MM_DD_YYYY_FORMAT)==CC_MM_DD_YYYY_FORMAT))
324 {
325 if(day<10) s.Append("0");
326 s<<day;
327
328 s.Append(div[divider]);
329
330 if(month<10) s.Append("0");
331 s<<month;
332 }
333 else // CC_MM_DD_YYYY_FORMAT
334 {
335 if(month<10) s.Append("0");
336 s<<month;
337
338 s.Append(div[divider]);
339
340 if(day<10) s.Append("0");
341 s<<day;
342 }
343 s.Append(div[divider]);
344
345 if((flags & CC_SHORT_YEAR)==CC_SHORT_YEAR)
346 {
347 int year1=year%100;
348 if(year1<10) s.Append("0");
349 s<<year1;
350 }
351 else // FULL_YEAR
352 {
353 if(year<10) s.Append("000");
354 else if(year<100) s.Append("00");
355 else if(year<1000) s.Append("0");
356 s<<year;
357 }
358
359 SetText(s.String());

364void DateTextView::GetDate(int *day, int *month, int *year)
365{
366 int mday=*day;
367 int mmonth=*month;
368 int myear=*year;
369
370 BString s(Text());
371 char n1[11];
372 char n2[11];
373 char n3[11];
374
375 s.CopyInto(n1,0,2);
376 n1[2]='\0';
377 s.CopyInto(n2,3,2);
378 n2[2]='\0';
379
380 if((flags & CC_SHORT_YEAR)==CC_SHORT_YEAR)
381 {
382 s.CopyInto(n3,6,2);
383 n3[2]='\0';
384 }
385 else // FULL_YEAR
386 {
387 s.CopyInto(n3,6,4);
388 n3[4]='\0';
389 }
390
391 if(!((flags & CC_MM_DD_YYYY_FORMAT)==CC_MM_DD_YYYY_FORMAT))
392 {
393 mday=atoi(n1);
394 mmonth=atoi(n2);
395 }
396 else
397 {
398 mday=atoi(n2);
399 mmonth=atoi(n1);
400 }
401
402 myear=atoi(n3);
403 if((flags & CC_SHORT_YEAR)==CC_SHORT_YEAR)
404 {
405 if((flags & CC_HALF_CENTURY)==CC_HALF_CENTURY)
406 {
407 if(myear<51) myear+=50; else myear-=50;
408 }
409 else if(myear==0) myear=100;
410
411 myear+=(first_year-1);
412 }
413
414 if(!VerifyDate(&mday,&mmonth,&myear)) SetDate(mday,mmonth,myear);
415
416 *day=mday;
417 *month=mmonth;
418 *year=myear;
419
420 return;
421}

436void DateTextView::SetDate(const char *tdate)
437{
438 // Almost the same as GetDate. May be to combine them.
439
440 // Changes text using current settings of control.
441 int day;
442 int month;
443 int year;
444
445 int k;
446
447 bool short_year=false;
448 if((flags & CC_SHORT_YEAR)==CC_SHORT_YEAR) short_year=true;
449
450 char n1[3]="00";
451 char n2[3]="00";
452 char n3[5]="0000";
453 if(short_year) n3[2]='\0';
454
455 bool zero=false; // was the end of tdate string?
456
457 int c=0;
458 while (!(c==2 || zero))
459 {
460 if(tdate[c]=='\0') zero=true;
461 else n1[c]=tdate[c];
462 c++;
463 }
464
465 if(zero) goto L1;
466 if(tdate[2]=='\0') goto L1;
467
468 c=0;
469 while (!(c==2 || zero))
470 {
471 if(tdate[c+3]=='\0') zero=true;
472 else n2[c]=tdate[c+3];
473 c++;
474 }
475
476 if(zero) goto L1;
477 if(tdate[5]=='\0') goto L1;
478
479 k=short_year ? 2 : 4;
480
481 c=0;
482 while (!(c==k || zero))
483 {
484 if(tdate[c+6]=='\0') zero=true;
485 else n3[c]=tdate[c+6];
486 c++;
487 }
488
489L1:
490 if(!((flags & CC_MM_DD_YYYY_FORMAT)==CC_MM_DD_YYYY_FORMAT))
491 {
492 day=atoi(n1);
493 month=atoi(n2);
494 }
495 else
496 {
497 day=atoi(n2);
498 month=atoi(n1);
499 }
500
501 year=atoi(n3);
502 if(short_year)
503 {
504 if((flags & CC_HALF_CENTURY)==CC_HALF_CENTURY)
505 {
506 if(year<51) year+=50; else year-=50;
507 }
508 else if(year==0) year=100;
509
510 year+=(first_year-1);
511 }
512 SetDate(day,month,year);
513}

517bool DateTextView::VerifyDate(int *day, int *month, int *year)
518{
519 // Function verifies date to be correct and changes it if it's needed
520 // Returns true if date was correct (and wasn't changed)
521
522 struct tm *dat;
523 time_t tmp;
524 time(&tmp);
525 dat=localtime(&tmp);
526
527 bool flag=true; // date is correct
528
529 if((flags & CC_SHORT_YEAR)==CC_SHORT_YEAR)
530 {
531 if(*year<first_year || *year>last_year)
532 {
533 int year1=*year%100;
534 if((flags & CC_HALF_CENTURY)==CC_HALF_CENTURY)
535 {
536 if(year1<51) year1+=50; else year1-=50;
537 }
538 else if(year1==0) year1=100;
539
540 *year=year1+first_year-1;
541 flag=false;
542 }
543 }
544 else // FULL_YEAR
545 {
546 if(*year<1 || *year>9999)
547 {
548 *year=dat->tm_year+1900;
549 flag=false;
550 }
551 }
552
553 if(*month<1 || *month>12)
554 {
555 *month=dat->tm_mon+1;
556 flag=false;
557 }
558
559 if(*day<1 || *day>31)
560 {
561 *day=dat->tm_mday;
562 flag=false;
563 }
564
565 if((*month==4 || *month==6 || *month==9 || *month==11) && *day>30)
566 {
567 if((*day=dat->tm_mday)>30) *day=30;
568 flag=false;
569 }
570 else if (*month==2)
571 {
572 int tmpday;
573
574 if((*year)%4==0) // leap year?
575 {
576 if((*year)%100==0 && (*year)%400!=0) tmpday=28; // no
577 else tmpday=29; // yes
578 }
579 else tmpday=28;
580
581 if(*day>tmpday)
582 {
583 if((*day=dat->tm_mday)>tmpday) *day=tmpday;
584 flag=false;
585 }
586 }
587
588 return flag;
589}
